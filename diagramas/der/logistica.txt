// =====================================================================
// DonaTrack - Servicio de Logística - Modelo de datos físico (MongoDB)
// Entrega 5: persistencia políglota. Ya no es un DER relacional (MySQL) --
// es un documento por colección, generado a partir de las clases @Document
// reales en domain/rutas y domain/eventos y de GeneradorIds/Secuencia.
// =====================================================================

Base de datos: logistica (MongoDB, replica set único para Logística)
IDs: enteros secuenciales (no ObjectId) asignados por GeneradorIds justo antes
del primer guardado de cada documento, para no romper el contrato de la API
REST, el broker con Donaciones ni los ejemplos de Postman/Swagger. El contador
de cada colección vive en secuencias (ver más abajo). EventoLogistico es la
única excepción: usa UUID propio, generado en su constructor, no por
GeneradorIds (no participa del contrato de ids enteros).

// ---------------------------------------------------------------------
// camiones
// ---------------------------------------------------------------------
{
  _id                  : Int               // GeneradorIds("camion")
  patente               : String            [index unique]
  capacidadVolumenM3    : Int
  alturaM               : Int
  capacidadCargaKg      : Int
  estadoCamion          : String            [index]  // DISPONIBLE | ASIGNADO | EN_RUTA | FUERA_DE_SERVICIO
}

// ---------------------------------------------------------------------
// choferes
// ---------------------------------------------------------------------
{
  _id         : Int                        // GeneradorIds("chofer")
  nombre      : String
  dni         : Int                         [index unique]
  habilitado  : Boolean                     [index]
}

// ---------------------------------------------------------------------
// rutas
// ---------------------------------------------------------------------
{
  _id             : Int                    // GeneradorIds("ruta")
  camionAsociado  : DBRef -> camiones       // referencia: el estado del camión se lee siempre actual, no se copia
  chofer          : DBRef -> choferes | null
  fecha           : Date                    [index]
  estadoRuta      : String                  [index]  // PLANIFICADA | INICIADA | FINALIZADA | CANCELADA
  paradas         : [ Parada ]              // EMBEBIDAS, en orden de visita -- la ruta se lee con una sola consulta
}

// Parada (documento EMBEBIDO dentro de rutas.paradas -- sin colección propia)
Parada {
  idParada                         : Int    // == numeroParada, no es un id global
  numeroParada                     : Int
  idEntidadBeneficiariaAsociada    : Int    // ref. lógica -> servicio Donaciones (EntidadBeneficiaria)
  direccion                        : Direccion   // embebido
  entregas                         : [ DBRef -> entregas ]  // referencias: cada entrega se consulta también por separado
}

// ---------------------------------------------------------------------
// entregas
// ---------------------------------------------------------------------
{
  _id                              : Int    // GeneradorIds("entrega")
  idDonacionAsociada                : Int    [index unique]  // ref. lógica -> Donacion; UNIQUE = idempotencia del lote
  idEntidadBeneficiariaAsociada     : Int    [index]          // ref. lógica -> EntidadBeneficiaria
  direccionDestino                 : Direccion   // embebido
  estadoEntrega                    : String  [index]  // PENDIENTE | ASIGNADA_A_RUTA | EN_TRASLADO | ENTREGADA | NO_RECIBIDA | FALLIDA | REPLANIFICABLE
  fecha                            : Date    [index]
  fechaHoraEntrega                 : DateTime | null   // se completa al confirmar (ENTREGADA)
  seguimientoUrl                   : String | null      // se completa al iniciar la ruta (RUTA_INICIADA)
  fotos                            : [ FotoEntrega ]     // EMBEBIDAS: no existen sin la entrega
  camionEntrega                    : DBRef -> camiones | null  // denormalizado a propósito respecto de rutas.camionAsociado
  pesoKG                           : Int
  volumenM3                        : Int
  alturaM                          : Int
  justificacionFallo               : String | null       // obligatoria si NO_RECIBIDA o FALLIDA
}

// Direccion (value object embebido, dentro de entregas y de Parada -- sin colección propia)
Direccion {
  calle    : String
  numero   : String
  ciudad   : Ciudad   // embebido
}
Ciudad {
  nombre     : String
  provincia  : Provincia  // embebido
}
Provincia {
  nombre : String
}

// FotoEntrega (documento embebido dentro de entregas.fotos -- sin colección propia)
FotoEntrega {
  url   : String
  fecha : Date
}

// ---------------------------------------------------------------------
// eventos_logisticos
// ---------------------------------------------------------------------
{
  _id               : UUID                  // generado en el constructor, no por GeneradorIds
  tipoEvento        : String                // RUTA_PLANIFICADA | RUTA_INICIADA | ENTREGA_EN_TRASLADO | ENTREGA_CONFIRMADA | ENTREGA_NO_RECIBIDA | ENTREGA_FALLIDA | ENTREGA_REPLANIFICADA
  publicado         : Boolean               [index]  // Donaciones consulta solo los no publicados
  entregaAsociada   : DBRef -> entregas      // Donaciones lee de ahí el id de donación y el camión
  fechaGeneracion   : DateTime              [index]

  // Bandeja de eventos de Logística; también funciona como historial de auditoría de cada entrega.
}

// ---------------------------------------------------------------------
// secuencias (soporte técnico de GeneradorIds -- no es dominio)
// ---------------------------------------------------------------------
{
  _id   : String   // "camion" | "chofer" | "entrega" | "ruta"
  valor : Long      // incrementado atómicamente con $inc + findAndModify antes de cada primer guardado
}
